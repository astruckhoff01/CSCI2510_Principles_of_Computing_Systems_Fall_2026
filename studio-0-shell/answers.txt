wc -l access.log
    - wc -l to count the lines

grep -c "404" access.log
    - grep: to find the status code ("404")
    - -c: to count the lines

awk '{print $1}' access.log | sort | uniq -c | sort -nr | head -n 3
    - awk '{print $1}': Extracts the first column (the IP address)
    - sort: Groups identical IP addresses together
    - uniq -c: Counts how many times each unique IP address appears
    - sort -nr: Sorts the list numerically (-n) in reverse order (-r), highest counts at the top
    - head -n 3: Limits the final output to just the top 3 

find . | sort -u access.log
    - find .: search current directory
    - sort -u: alphabetically sorts and removes duplicates

awk '{print $2}' access.log | sort | uniq -c | sort -rn | awk '{print $2, $1}'
    - awk '{print $2}': the status codes
    - sort: sorts alphabetically
    - uniq -c: counts the occurences of status codes
    - sort -rn: sorts numerically in reverse order
    - awk '{print $2, $1}': swaps to 'status count' order
